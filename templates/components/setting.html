{#- Resolves a theme setting: page > nearest ancestor section > current section > config > default. -#} {%- component terminus.setting(setting: string, default_global_value = false, @page = none, @section = none, @config) -%} {%- set ancestor_setting = none -%} {%- if page and page.ancestors -%} {%- for ancestor in page.ancestors | reverse -%} {%- set section_data = get_section(path=ancestor, metadata_only=true) -%} {%- if setting in section_data.extra -%} {%- set_global ancestor_setting = section_data.extra[setting] -%} {%- break -%} {%- endif -%} {%- endfor -%} {%- endif -%} {{- page?.extra?[setting] or section?.extra?[setting] or ancestor_setting or config.extra?[setting] or default_global_value -}} {%- endcomponent terminus.setting -%}
